\begin{tikzpicture}[
  blk/.style   = {draw, thick, rounded corners=2pt, align=center,
                  font=\small, minimum height=10mm, minimum width=24mm},
  pwr/.style   = {very thick, -{Latex[length=2.2mm]}},
  bus/.style   = {thick, dashed, -{Latex[length=2.2mm]}},
  ser/.style   = {thick, dotted, -{Latex[length=2.2mm]}},
  gnd/.style   = {semithick, gray!70},
  lbl/.style   = {font=\scriptsize, inner sep=1.5pt},
  note/.style  = {font=\scriptsize\itshape, inner sep=2pt},
  pcap/.style  = {font=\small, align=center},
]

% =====================  PANEL (a): BENCH  =====================
\begin{scope}
  \node[blk] (src)  at (0,0)     {eZ-FET 3V3\\[-1pt]\scriptsize source};
  \node[blk] (ina)  at (4.0,0)   {INA226\\[-1pt]\scriptsize $R_{\mathrm{sh}}=0.1\,\Omega$\\[-2pt]\scriptsize VBS\,$\to$\,IN$-$};
  \node[blk] (mcu)  at (8.4,0)   {MSP430G2553\\[-1pt]\scriptsize target VCC, 16\,MHz};
  \node[blk] (ard)  at (4.0,-2.1){Arduino Uno\\[-1pt]\scriptsize INA226 reader};
  \node[blk] (pc)   at (8.4,-2.1){Host PC};

  % power path
  \draw[pwr] (src) -- node[lbl,above]{IN$+$} (ina);
  \draw[pwr] (ina) -- node[lbl,above]{IN$-$}
                      node[note,below=1pt]{measured rail} (mcu);

  % INA226 read-out + host links
  \draw[bus] (ina) -- node[lbl,align=left,right=2pt]{I\textsuperscript{2}C (A4/A5)\\+ VCC 5\,V} (ard);
  \draw[ser] (ard) -- node[lbl,above]{CSV, 115.2\,kBd} (pc);
  \draw[ser] (mcu) -- node[lbl,right]{UART, 9.6\,kBd} (pc);

  % common ground
  \draw[gnd] (-1.4,-3.3) -- (9.8,-3.3);
  \foreach \n in {src,ina,mcu,ard} { \draw[gnd] (\n.south) -- (\n.south |- 0,-3.3); }
  \node[lbl,gray!70] at (-1.4,-3.55) {GND};

  % measurement annotations
  \node[note, anchor=west] at (4.9,0.95) {system power $P=V\!\cdot\!I$ measured here};
  \draw[gray,-{Latex[length=1.6mm]}] (4.9,0.85) -- (4.15,0.32);
  \node[note, anchor=west] at (8.4,1.25) {latency timed in firmware};
  \draw[gray,-{Latex[length=1.6mm]}] (8.9,1.12) -- (8.6,0.55);

  \node[pcap] at (4.2,-4.15) {\textbf{(a)} Bench, MCU-only (\texttt{TEST\_MODE\,3}) --- embedded data, \emph{sensor not connected}};
\end{scope}

% =====================  PANEL (b): SENSOR IN THE LOOP  =====================
\begin{scope}[yshift=-6.6cm]
  \node[blk] (src2) at (0,0)     {eZ-FET 3V3\\[-1pt]\scriptsize source};
  \node[blk] (ina2) at (4.0,0)   {INA226\\[-1pt]\scriptsize $R_{\mathrm{sh}}=0.1\,\Omega$\\[-2pt]\scriptsize VBS\,$\to$\,IN$-$};
  \node[blk] (mcu2) at (8.4,0)   {MSP430G2553\\[-1pt]\scriptsize target VCC, 16\,MHz};
  \node[blk] (ard2) at (4.0,-2.1){Arduino Uno\\[-1pt]\scriptsize INA226 reader};
  \node[blk] (mpu)  at (8.4,-2.1){MPU6050\\[-1pt]\scriptsize accelerometer};

  % power path
  \draw[pwr] (src2) -- node[lbl,above]{IN$+$} (ina2);
  \draw[pwr] (ina2) -- node[lbl,above]{IN$-$}
                       node[note,below=1pt]{measured rail} (mcu2);

  % sensor powered FROM the measured rail (key methodological point)
  \draw[pwr] (6.2,0) |- node[lbl,pos=0.72,above]{$V_{\mathrm{CC}}$} (mpu.west);

  % sensor I2C to the MCU
  \draw[bus] (mcu2) -- node[lbl,align=left,right=2pt]{I\textsuperscript{2}C\\P1.6\,/\,P1.7} (mpu);

  % INA226 read-out
  \draw[bus] (ina2) -- node[lbl,align=left,right=2pt]{I\textsuperscript{2}C (A4/A5)\\+ VCC 5\,V} (ard2);

  % common ground
  \draw[gnd] (-1.4,-3.3) -- (9.8,-3.3);
  \foreach \n in {src2,ina2,mcu2,ard2,mpu} { \draw[gnd] (\n.south) -- (\n.south |- 0,-3.3); }
  \node[lbl,gray!70] at (-1.4,-3.55) {GND};
  \node[lbl,gray!70,anchor=west] at (9.75,-2.1) {AD0$\to$GND\;(0x68)};

  \node[pcap] at (4.2,-4.15) {\textbf{(b)} Sensor-in-the-loop --- identical wiring for\\[-2pt]\texttt{TEST\_MODE\,4} (latency) and \texttt{TEST\_MODE\,5} (energy)};
\end{scope}
